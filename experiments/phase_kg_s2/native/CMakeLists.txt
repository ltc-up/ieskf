cmake_minimum_required(VERSION 3.16)
project(kg_s2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(GTSAM REQUIRED)
find_package(GTSAM_UNSTABLE REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(kg_s2 joint.cpp)
target_include_directories(kg_s2 PRIVATE ${POINTS_SOURCE}/include ${POINTS_BUILD}/include /opt/ros/jazzy/include)
target_link_libraries(kg_s2 PRIVATE ${POINTS_BUILD}/libgtsam_points.so gtsam gtsam_unstable Eigen3::Eigen)
set_target_properties(kg_s2 PROPERTIES BUILD_RPATH "${POINTS_BUILD};/opt/ros/jazzy/lib/x86_64-linux-gnu")
add_executable(kg_diag diagnostics.cpp)
target_include_directories(kg_diag PRIVATE ${POINTS_SOURCE}/include ${POINTS_BUILD}/include /opt/ros/jazzy/include)
target_link_libraries(kg_diag PRIVATE ${POINTS_BUILD}/libgtsam_points.so gtsam gtsam_unstable Eigen3::Eigen)
set_target_properties(kg_diag PROPERTIES BUILD_RPATH "${POINTS_BUILD};/opt/ros/jazzy/lib/x86_64-linux-gnu")
add_executable(kg_tests tests.cpp)
target_include_directories(kg_tests PRIVATE ${POINTS_SOURCE}/include ${POINTS_BUILD}/include /opt/ros/jazzy/include)
target_link_libraries(kg_tests PRIVATE ${POINTS_BUILD}/libgtsam_points.so gtsam gtsam_unstable Eigen3::Eigen)
set_target_properties(kg_tests PROPERTIES BUILD_RPATH "${POINTS_BUILD};/opt/ros/jazzy/lib/x86_64-linux-gnu")
